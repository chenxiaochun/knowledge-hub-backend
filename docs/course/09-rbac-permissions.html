目标与验收

前置：第 2 课 JWT + RolesGuard 已通。

本课：把 role_codes 字符串列升级为多表 RBAC；接口可用 @RequirePermission 做权限码鉴权。

验收（两个非 ADMIN 用户，同角色不同权限）：
  1. 种子角色 EDITOR + 权限 system:user / document:list
  2. 用户 alice 只有 document:list；bob 有 system:user
  3. alice 调 GET /users/page → 403；bob → 200
  4. ADMIN 任意权限码接口 → 200（守卫短路）
# 用 admin 登录后给用户绑角色/权限（见下文 API）
# 再分别用 alice / bob 的 token：
curl -s -o /dev/null -w "%{http_code}" http://localhost:3000/users/page \
  -H "Authorization: Bearer $ALICE_TOKEN"
# 期望 403
curl -s -o /dev/null -w "%{http_code}" http://localhost:3000/users/page \
  -H "Authorization: Bearer $BOB_TOKEN"
# 期望 200

概念

课 2本课

  3. 装饰器 + PermissionsGuard

    src/auth/decorators/require-permission.decorator.ts

    import { SetMetadata } from '@nestjs/common';

export const PERMISSIONS_KEY = 'permissions';

/** 拥有其中任一权限码即可；需配合 PermissionsGuard；ROLE_ADMIN 自动放行 */
export const RequirePermission = (...permissions: string[]) =>
  SetMetadata(PERMISSIONS_KEY, permissions);

    src/auth/permissions.guard.ts

    import {
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Injectable,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { AuthUser } from './auth-user.interface';
import { PERMISSIONS_KEY } from './decorators/require-permission.decorator';
import { RoleCode } from '../common/constants/roles';

@Injectable()
export class PermissionsGuard implements CanActivate {
  constructor(private readonly reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    const required = this.reflector.getAllAndOverride<string[]>(
      PERMISSIONS_KEY,
      [context.getHandler(), context.getClass()],
    );
    if (!required?.length) return true;

    const request = context.switchToHttp().getRequest<{ user?: AuthUser }>();
    const user = request.user;
    if (!user) throw new ForbiddenException('权限不足');

    if (user.roles.includes(RoleCode.ADMIN)) return true;

    const owned = new Set(user.permissions ?? []);
    if (!required.some((p) => owned.has(p))) {
      throw new ForbiddenException('权限不足');
    }
    return true;
  }
}

  5. RbacService：聚合权限 + 绑角色

    src/user/rbac.service.ts

    import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { In, Repository } from 'typeorm';
import { nextSnowflakeId } from '../common/snowflake-id';
import { RoleEntity } from './entities/role.entity';
import { PermissionEntity } from './entities/permission.entity';
import { UserRoleEntity } from './entities/user-role.entity';
import { RolePermissionEntity } from './entities/role-permission.entity';
import { UserEntity } from './entities/user.entity';

@Injectable()
export class RbacService {
  constructor(
    @InjectRepository(RoleEntity)
    private readonly roleRepo: Repository<RoleEntity>,
    @InjectRepository(PermissionEntity)
    private readonly permRepo: Repository<PermissionEntity>,
    @InjectRepository(UserRoleEntity)
    private readonly userRoleRepo: Repository<UserRoleEntity>,
    @InjectRepository(RolePermissionEntity)
    private readonly rolePermRepo: Repository<RolePermissionEntity>,
    @InjectRepository(UserEntity)
    private readonly userRepo: Repository<UserEntity>,
  ) {}

  async listRoles() {
    return this.roleRepo.find({ where: { status: 1 } });
  }

  async listPermissions() {
    return this.permRepo.find({ where: { status: 1 } });
  }

  /** 用户的角色码 + 权限码（供 buildAuthUser） */
  async loadRolesAndPermissions(userId: string): Promise<{
    roles: string[];
    permissions: string[];
  }> {
    const urs = await this.userRoleRepo.find({ where: { userId } });
    if (!urs.length) return { roles: [], permissions: [] };

    const roleIds = urs.map((u) => u.roleId);
    const roles = await this.roleRepo.find({ where: { id: In(roleIds) } });
    const roleCodes = roles.map((r) => r.roleCode);

    const rps = await this.rolePermRepo.find({ where: { roleId: In(roleIds) } });
    if (!rps.length) return { roles: roleCodes, permissions: [] };

    const perms = await this.permRepo.find({
      where: { id: In(rps.map((p) => p.permissionId)) },
    });
    return {
      roles: roleCodes,
      permissions: [...new Set(perms.map((p) => p.permissionCode))],
    };
  }

  /** 覆盖式绑定：先删旧关联再插入 */
  async setUserRoles(userId: string, roleCodes: string[]) {
    const user = await this.userRepo.findOne({
      where: { id: userId, deleted: false },
    });
    if (!user) throw new NotFoundException('用户不存在');

    const roles = await this.roleRepo.find({
      where: { roleCode: In(roleCodes) },
    });
    if (roles.length !== roleCodes.length) {
      throw new NotFoundException('存在未知 roleCode');
    }

    await this.userRoleRepo.delete({ userId });
    const rows = roles.map((r) =>
      this.userRoleRepo.create({
        id: nextSnowflakeId(),
        userId,
        roleId: r.id,
      }),
    );
    await this.userRoleRepo.save(rows);
    return this.loadRolesAndPermissions(userId);
  }

  async setRolePermissions(roleCode: string, permissionCodes: string[]) {
    const role = await this.roleRepo.findOne({ where: { roleCode } });
    if (!role) throw new NotFoundException('角色不存在');

    const perms = await this.permRepo.find({
      where: { permissionCode: In(permissionCodes) },
    });
    if (perms.length !== permissionCodes.length) {
      throw new NotFoundException('存在未知 permissionCode');
    }

    await this.rolePermRepo.delete({ roleId: role.id });
    const rows = perms.map((p) =>
      this.rolePermRepo.create({
        id: nextSnowflakeId(),
        roleId: role.id,
        permissionId: p.id,
      }),
    );
    await this.rolePermRepo.save(rows);
    return { roleCode, permissionCodes };
  }
}

  6. 启动种子

    src/user/rbac-seed.service.ts

    import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { nextSnowflakeId } from '../common/snowflake-id';
import { RoleCode } from '../common/constants/roles';
import { PermissionCode } from '../common/constants/permissions';
import { RoleEntity } from './entities/role.entity';
import { PermissionEntity } from './entities/permission.entity';
import { RolePermissionEntity } from './entities/role-permission.entity';

@Injectable()
export class RbacSeedService implements OnModuleInit {
  private readonly logger = new Logger(RbacSeedService.name);

  constructor(
    @InjectRepository(RoleEntity)
    private readonly roleRepo: Repository<RoleEntity>,
    @InjectRepository(PermissionEntity)
    private readonly permRepo: Repository<PermissionEntity>,
    @InjectRepository(RolePermissionEntity)
    private readonly rolePermRepo: Repository<RolePermissionEntity>,
  ) {}

  async onModuleInit() {
    await this.ensureRoles();
    await this.ensurePermissions();
    await this.ensureEditorDefaults();
    this.logger.log('RBAC 种子已就绪');
  }

  private async ensureRoles() {
    const defs = [
      { code: RoleCode.ADMIN, name: '管理员' },
      { code: RoleCode.EDITOR, name: '编辑' },
      { code: RoleCode.VIEWER, name: '访客' },
      { code: RoleCode.USER, name: '普通用户' },
    ];
    for (const d of defs) {
      const exists = await this.roleRepo.findOne({ where: { roleCode: d.code } });
      if (exists) continue;
      await this.roleRepo.save(
        this.roleRepo.create({
          id: nextSnowflakeId(),
          roleCode: d.code,
          roleName: d.name,
          status: 1,
        }),
      );
    }
  }

  private async ensurePermissions() {
    const defs = [
      { code: PermissionCode.systemUser, name: '用户管理' },
      { code: PermissionCode.documentList, name: '文档列表' },
      { code: PermissionCode.documentCreate, name: '文档创建' },
      { code: PermissionCode.documentReview, name: '文档审核' },
      { code: PermissionCode.search, name: '搜索' },
    ];
    for (const d of defs) {
      const exists = await this.permRepo.findOne({
        where: { permissionCode: d.code },
      });
      if (exists) continue;
      await this.permRepo.save(
        this.permRepo.create({
          id: nextSnowflakeId(),
          permissionCode: d.code,
          permissionName: d.name,
          permissionType: 3,
          status: 1,
        }),
      );
    }
  }

  /** EDITOR 默认只有 document:list（验收时再给个别用户加 system:user） */
  private async ensureEditorDefaults() {
    const editor = await this.roleRepo.findOne({
      where: { roleCode: RoleCode.EDITOR },
    });
    if (!editor) return;
    const count = await this.rolePermRepo.count({ where: { roleId: editor.id } });
    if (count > 0) return;

    const list = await this.permRepo.findOne({
      where: { permissionCode: PermissionCode.documentList },
    });
    if (!list) return;
    await this.rolePermRepo.save(
      this.rolePermRepo.create({
        id: nextSnowflakeId(),
        roleId: editor.id,
        permissionId: list.id,
      }),
    );
  }
}
    若课 2 的 RoleCode 没有 EDITOR / VIEWER，在 roles.ts 补上即可。

  7. 改 buildAuthUser：读多表，不再只信 role_codes

    在 UserService.buildAuthUser（或 AuthService 等价方法）里：

    async buildAuthUser(user: UserEntity): Promise<AuthUser> {
  const { roles, permissions } =
    await this.rbacService.loadRolesAndPermissions(user.id);

  // 迁移期：若还没有任何 user_role，可回退读 role_codes（可选）
  const fallbackRoles =
    roles.length > 0
      ? roles
      : (user.roleCodes ?? '')
          .split(',')
          .map((s) => s.trim())
          .filter(Boolean);

  return {
    userId: user.id,
    username: user.username,
    realName: user.realName,
    email: user.email,
    roles: fallbackRoles,
    permissions,
  };
}

    # 1) admin 登录拿 TOKEN
# 2) 确保 EDITOR 角色权限：
curl -s -X PUT http://localhost:3000/users/roles/ROLE_EDITOR/permissions \
  -H "Authorization: Bearer $ADMIN" -H 'Content-Type: application/json' \

curl -s -X PUT http://localhost:3000/users/roles/ROLE_EDITOR/permissions \
  -H "Authorization: Bearer $ADMIN" -H 'Content-Type: application/json' \
  -d '{"permissionCodes":["document:list","system:user"]}'

# 「同角色不同权限」更干净的做法：两个角色 ROLE_A / ROLE_B，权限不同，用户各绑一个。

curl -s -X PUT http://localhost:3000/users/$ALICE_ID/roles \
  -H "Authorization: Bearer $ADMIN" -H 'Content-Type: application/json' \
  -d '{"roleCodes":["ROLE_EDITOR"]}'
# EDITOR 仅 document:list 时 alice 访问 /users/page → 403

设计决策

为什么权限挂在角色上，而不是直接挂用户？

角色是权限打包；改一处角色权限，所有成员立刻生效。origin 另有用户直接权限表，本课先不做，避免两套来源难排查。

为什么 ADMIN 短路放行？

与 origin PermissionsGuard 一致，避免给管理员维护几百条权限码。生产若要「超级管理员也审计」，再关掉短路。
